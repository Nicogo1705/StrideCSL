namespace Demo
{
    /// One level of the occupancy pyramid from the level under it: each brick is the union of its
    /// eight children, which the one-sample overlap makes exact. Only for the bricks whose samples
    /// touch an active brick.
    shader VoxelWaterOccupancyUp : ComputeShaderBase, VoxelWaterBricks
    {
        /// Single-mip views: the finer level to read, this level to write.
        stage Texture3D<float2> Source;
        stage RWTexture3D<float2> Target;
        stage int3 SourceSize;
        stage int3 TargetSize;

        /// Cells along a brick's edge at this level: 2^(mip + 1).
        stage int BrickCells;

        override void Compute()
        {
            int3 b = int3(streams.DispatchThreadId);
            if (any(b >= TargetSize))
                return;
            int3 first = b * BrickCells;
            if (!BoxDirty(first, first + BrickCells))
                return;

            float low = 1.0;
            float high = 0.0;
            [unroll]
            for (int z = 0; z <= 1; z++)
            {
                [unroll]
                for (int y = 0; y <= 1; y++)
                {
                    [unroll]
                    for (int x = 0; x <= 1; x++)
                    {
                        int3 child = min(b * 2 + int3(x, y, z), SourceSize - 1);
                        float2 value = Source.Load(int4(child, 0));
                        low = min(low, value.r);
                        high = max(high, value.g);
                    }
                }
            }
            Target[b] = float2(low, high);
        }
    };
}
